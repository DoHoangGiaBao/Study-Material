int y = 42;									->			addi x2, x0, 42
int max;									->			addi x3, x0, 0

if (x >= y) {								->			lt x1, x2, ELSE
    max = x;											add x3, x0, x1
} else {												j END	 
    max = y;									ELSE:	add x3, x0, x2
}												END:
(x->x1, y->x2, max->x3)

int sum = 0;								->			addi x2, x0, 0
														addi x3, x0, 5 # Store number 5
while (i < 5) {								->	LOOP:	bge x1, x3, END
    sum = sum + i;										add x2, x2, x1
    i++;												addi x1, x1, 1
}														j LOOP
												END:
(i->x1, sum->x2, 5->x3)
